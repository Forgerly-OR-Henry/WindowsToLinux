cmake_minimum_required(VERSION 3.20)
project(log-worker LANGUAGES CXX)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(log-worker src/main.cpp src/worker.cpp src/log.cpp)
target_include_directories(log-worker PRIVATE include)
target_include_directories(log-worker SYSTEM PRIVATE vendor)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(log-worker PRIVATE -Wall -Wextra -Werror)
endif()
if(MINGW)
 target_link_options(log-worker PRIVATE -municode -static-libgcc)
 if("CXX" STREQUAL "CXX")
  target_link_options(log-worker PRIVATE -static-libstdc++)
 endif()
endif()
