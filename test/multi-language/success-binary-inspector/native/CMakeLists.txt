cmake_minimum_required(VERSION 3.20)
project(binary-worker LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(binary-worker src/main.c src/worker.c src/format.c)
target_include_directories(binary-worker PRIVATE include)
target_include_directories(binary-worker SYSTEM PRIVATE vendor)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang" OR CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(binary-worker PRIVATE -Wall -Wextra -Werror)
endif()
if(MINGW)
 target_link_options(binary-worker PRIVATE -municode -static-libgcc)
 if("C" STREQUAL "CXX")
  target_link_options(binary-worker PRIVATE -static-libstdc++)
 endif()
endif()
